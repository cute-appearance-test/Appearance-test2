  <script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>
  <script>
    // ----------------------------------------------------
    // Supabase Configuration
    // ----------------------------------------------------
    const SUPABASE_URL = 'https://fcnuoymkjnpksqedodsb.supabase.co';
    const SUPABASE_ANON_KEY = 'YOUR_SUPABASE_ANON_KEY'; //sb_publishable_PkfnJEQtRh8RApjDSJh_zA_efiQCc1A
    const _supabase = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    // DOM Elements
    const webcam = document.getElementById('webcam');
    const canvas = document.getElementById('canvas');
    const snapBtn = document.getElementById('snapBtn');
    const stepIndicator = document.getElementById('stepIndicator');
    const resultsCard = document.getElementById('resultsCard');
    const pingDisplay = document.getElementById('pingDisplay');
    
    const prevFront = document.getElementById('prevFront');
    const prevLeft = document.getElementById('prevLeft');
    const prevRight = document.getElementById('prevRight');

    let currentStep = 0;
    const stepTitles = [
      "📸 ステップ 1/3：正面のお顔",
      "📸 ステップ 2/3：左側のお顔",
      "📸 ステップ 3/3：右側のお顔"
    ];

    let capturedBlobs = [];
    let capturedDataURLs = [];

    // Initialize Camera
    async function initCamera() {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" } });
        webcam.srcObject = stream;
      } catch (err) {
        console.error("カメラのアクセスに失敗しました:", err);
        alert("カメラへのアクセスを許可してください。");
      }
    }

    // Step Capture Function
    async function takeStepPhoto() {
      const context = canvas.getContext('2d');
      canvas.width = webcam.videoWidth || 320;
      canvas.height = webcam.videoHeight || 240;
      context.drawImage(webcam, 0, 0, canvas.width, canvas.height);

      const dataURL = canvas.toDataURL('image/png');
      capturedDataURLs.push(dataURL);

      await new Promise((resolve) => {
        canvas.toBlob((blob) => {
          capturedBlobs.push(blob);
          resolve();
        }, 'image/png');
      });

      currentStep++;

      if (currentStep < 3) {
        stepIndicator.innerText = stepTitles[currentStep];
        alert(撮影完了！次は【${stepTitles[currentStep].replace(/📸 ステップ \d\/\d：/, '')}】を撮影します。);
      } else {
        snapBtn.disabled = true;
        snapBtn.innerText = "🔄 測定中 ＆ アップロード中...";
        processAndUpload();
      }
    }

    // Ping Measurement & Supabase Upload
    async function processAndUpload() {
      // Show Preview Images
      prevFront.src = capturedDataURLs[0];
      prevLeft.src = capturedDataURLs[1];
      prevRight.src = capturedDataURLs[2];
      resultsCard.style.display = 'block';

      // 1. Measure Latency / Ping
      const startTime = performance.now();
      try {
        await fetch(${SUPABASE_URL}/rest/v1/, { method: 'HEAD' });
      } catch (e) {
        console.log("Ping measure fallback");
      }
      const endTime = performance.now();
      const pingMs = Math.round(endTime - startTime);

      pingDisplay.innerText = 応答速度（レイテンシー）: ${pingMs} ms;

      // 2. Upload Photos & Save Record
      try {
        const timestamp = Date.now();
        const uploadedUrls = [];

        const angles = ['front', 'left', 'right'];
        for (let i = 0; i < 3; i++) {
          const fileName = delay_${angles[i]}_${timestamp}.png;
          const { error: uploadError } = await _supabase.storage
            .from('delay-photos')
            .upload(fileName, capturedBlobs[i]);

          if (uploadError) throw uploadError;

          const { data: { publicUrl } } = _supabase.storage
            .from('delay-photos')
            .getPublicUrl(fileName);

          uploadedUrls.push(publicUrl);
        }

        // 3. Insert record into database
        const { error: dbError } = await _supabase
          .from('delay_tests')
          .insert([
            {
              photo_front: uploadedUrls[0],
              photo_left: uploadedUrls[1],
              photo_right: uploadedUrls[2],
              ping_ms: pingMs
            }
          ]);

        if (dbError) throw dbError;

        snapBtn.innerText = "✨ 送信が完了しました！";
        stepIndicator.innerText = "🎉 すべてのステップが完了しました";

      } catch (err) {
        console.error("アップロードエラー:", err);
        alert("データの送信に失敗しました: " + err.message);
        snapBtn.disabled = false;
        snapBtn.innerText = "再試行する 🍃";
      }
    }

    window.addEventListener('DOMContentLoaded', initCamera);
  </script>
